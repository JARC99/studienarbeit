	\section{Einleitung \& Motivation}
		\begin{frame}{Einleitung \& Motivation}
			\begin{itemize}
				\item Die immer noch weit verbreite Verwendung von Gasturbinen in verschiedenen Bereichen des menschliches Leben erfordert die Weiterentwicklung dieser Systeme mit der Ziel, ihre Betriebskosten und Umweltbelastung soweit wie möglich zu reduzieren.
				
				\item Die Minimierung von sekundären Strömungen und Mischverluste innerhalb der Maschine bietet immer noch eine gute Gelegenheit, den thermischen Wirkungsgrad weiter zu erhöhen \cite{farokhi2014}.
				

			\end{itemize}
		\end{frame}
		
		\begin{frame}{Einleitung \& Motivation}
			\begin{itemize}
				
				\item In \cite{palkus2023low} wurden zwei Lösungen entwickelt, um sekundär Strömungen innerhalb den OAS von LPT zu minimieren: Front Deflectors und Vane Bleed Holes (VBH).
				
				\item Von diesen zwei haben sich, explorativen numerischen Studien nach, VBH bezüglich der möglichen Effizienzsteigerung  als mehr versprechend erwiesen.
				
				\item Die Wirkungsgrdverbesserungen mittels des VBH-Konzepts wurde in \cite{camareropueyo2025reducing,camareropueyo2026comparison,camareropueyo2026robustness} mittels numerischer Optimizationstudien als Verbesserungsmöglichkeit für LPT weiterrecherchiert und bestätigt.
				
				\item Analog wurde die Optimierung der Wandkontouren der Leitschaufeln einer HDT in [CITA] durchgeführt, was auch eine gewisse Effizienzstereigerung ermöglicht hat.
			\end{itemize}
		\end{frame}
		
		\begin{frame}{Einleitung \& Motivation}
			\begin{itemize}
				\item Das Ziel dieser Studienarbeit ist dann die numerische Untersuchung von den Effekten der Umsetzung von VBH in einer Maschine mit voher optimierter Wandkontouren.
				
				\item Das Endergebnis von [CITA] wurde als Referenzsetup benutzt. Drei verschiedene VBH-Fälle wurden manuell gestaltet und in allen möglichen Permutation simuliert.
			\end{itemize}
		\end{frame}
