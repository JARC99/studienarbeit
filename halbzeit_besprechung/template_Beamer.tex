	\section{Einleitung \& Motivation}
	\begin{frame}{Einleitung \& Motivation}
		\begin{itemize}
			\item Die immer noch weit verbreitete Verwendung von Gasturbinen in verschiedenen Bereichen des menschlichen Lebens erfordert die Weiterentwicklung dieser Systeme mit dem Ziel, ihre Betriebskosten und Umweltbelastung so weit wie möglich zu reduzieren.
			
			\item Die Minimierung von Sekundärströmungen und Mischverlusten innerhalb der Maschine bietet immer noch eine gute Gelegenheit, den thermischen Wirkungsgrad weiter zu erhöhen \cite{farokhi2014}.
		\end{itemize}
	\end{frame}
	
	\begin{frame}{Einleitung \& Motivation}
		\begin{itemize}
			\item In \cite{palkus2023low} wurden zwei Lösungen entwickelt, um Sekundärströmungen innerhalb der Leitschaufelreihen (OAS) von Niederdruckturbinen (LPT) zu minimieren: Front Deflectors und Vane Bleed Holes (VBH).
			
			\item Von diesen beiden haben sich explorativen numerischen Studien zufolge VBH bezüglich der möglichen Effizienzsteigerung als vielversprechender erwiesen.
			
			\item Die Wirkungsgradverbesserungen mittels des VBH-Konzepts wurden in \cite{camareropueyo2025reducing,camareropueyo2026comparison,camareropueyo2026robustness} mithilfe numerischer Optimierungsstudien als Verbesserungsmöglichkeit für LPT weiter untersucht und bestätigt.
			
			\item Analog wurde die Optimierung der Wandkonturen der Leitschaufeln einer Hochdruckturbine (HDT) in \cite{Bittner2020} durchgeführt, was ebenfalls eine gewisse Effizienzsteigerung ermöglicht hat.
		\end{itemize}
	\end{frame}
	
	\begin{frame}{Einleitung \& Motivation}
		\begin{itemize}
			\item Das Ziel dieser Studienarbeit ist die numerische Untersuchung der Effekte der Umsetzung von VBH in einer Maschine mit vorher optimierten Wandkonturen.
			
			\item Das Endergebnis aus \cite{Bittner2020} wurde als Referenzsetup genutzt. Drei verschiedene VBH-Fälle wurden manuell gestaltet und in allen möglichen Permutationen simuliert.
		\end{itemize}
	\end{frame}

	\subsection{Entwickelte VBH Konstruktionen}
	\begin{frame}{Entwickelte VBH Konstruktionen}
		\begin{itemize}
			\item Drei verschiedene VBH-Konstruktionen wurden entwickelt und simuliert. 
			\item Laut \cite{palkus2023low} können Änderungen der Positionierung der Austrittsfläche der VBH entlang der Profilsehne zu Änderungen der Effizienzsteigerung führen. 
			\item Demnach wurden zwei unterschiedliche Austrittsflächen für alle Fälle erprobt: am Throat des Gitters und am Inflektionspunkt.
			\item Darüber hinaus wurde, um ein besseres Druckverhältnis schaffen zu können, eine dritte Konstruktion mit einer Eintrittsfläche entwickelt, die mit der Mitte der Kavität verbunden ist. 
		\end{itemize}
	\end{frame}
